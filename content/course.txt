# ============================================================
LESSON 1 1 First programs
SUMMARY main, printf, format specifiers and scanf
# ------------------------------------------------------------
CARD Your first C program
TEXT Every C program starts running at the `main()` function. The line `#include <stdio.h>` gives us the input and output functions, such as `printf`.
TEXT Every statement ends with a semicolon `;` and blocks of code sit inside curly braces `{ }`. C is compiled: your code is turned into machine code before it runs.
TEXT `return 0;` at the end of `main` tells the system that everything worked.
CODE #include <stdio.h>   // gives us printf and scanf
CODE
CODE int main() {
CODE     printf("Hello, World!\n");
CODE     return 0;   // 0 means "everything worked"
CODE }
END
CARD Printing values with printf
TEXT `printf` prints text, and format specifiers say where values go: `%d` for an int, `%f` for a float or double, `%c` for a char and `%s` for a string.
TEXT `%.2f` prints a number with 2 decimal places, and `\n` starts a new line.
CODE int age = 25;
CODE float price = 19.99f;
CODE char grade = 'A';
CODE
CODE printf("Age: %d\n", age);         // Age: 25
CODE printf("Price: %.2f\n", price);   // Price: 19.99
CODE printf("Grade: %c\n", grade);     // Grade: A
END
CARD Reading input with scanf
TEXT `scanf` reads what the user types and stores it in a variable. It needs the variable's address, which is what the `&` means: "address of".
TEXT When reading a single character, put a space before `%c` (as in `" %c"`) so it skips any leftover spaces or new lines.
CODE int age;
CODE char initial;
CODE
CODE printf("Enter your age: ");
CODE scanf("%d", &age);        // & = the address of age
CODE
CODE printf("Enter your initial: ");
CODE scanf(" %c", &initial);   // space before %c skips whitespace
END

CODE printf("Age: %d, Price: %.2f\n", age, price);

EXPLAIN %d is replaced by age (25), and %.2f prints price with exactly 2 decimal places: 19.99. Plain %f would print 19.990000.
END

LESSON 2 1 Variables and maths
SUMMARY Types, const, integer division, remainder and casting
# ------------------------------------------------------------
CARD Variables and types
TEXT In C you must declare the type of every variable before using it. `int` holds whole numbers (usually 4 bytes), `float` and `double` hold decimals (4 and 8 bytes), and `char` holds a single character in single quotes.
TEXT A variable that isn't given a value isn't set to 0: it holds whatever value was already in that memory.
TEXT `const` makes a variable unchangeable. Trying to change it is a compile error.
CODE int var;              // not initialised: whatever was in memory
CODE int age = 25;         // whole number
CODE double pi = 3.14159;  // decimal (8 bytes)
CODE char grade = 'A';     // one character, single quotes
CODE
CODE const int MAX_SCORE = 100;
CODE // MAX_SCORE = 120;   // error: it's const
END
CARD Doing maths
TEXT With two ints, `/` does integer division: the decimal part is cut off, so `10 / 3` is 3. The `%` operator gives the remainder: `10 % 3` is 1.
TEXT `count++` adds 1 and `count--` takes 1 away. `num += 5` is short for `num = num + 5`, and `num *= 2` for `num = num * 2`.
CODE int a = 10, b = 3;
CODE printf("%d\n", a / b);   // 3 (integer division)
CODE printf("%d\n", a % b);   // 1 (remainder)
CODE
CODE int num = 10;
CODE num += 5;    // 15
CODE num *= 2;    // 30
END
CARD Casting
TEXT Casting converts a value to another type. `(float)7 / 2` turns 7 into a float first, so the division keeps its decimal: 3.5.
TEXT Converting a float to an int cuts off the decimal part rather than rounding, so `(int)3.9` is 3.
CODE int result = 7 / 2;            // 3 (truncated)
CODE float exact = (float)7 / 2;    // 3.5
CODE int cut = (int)3.9;            // 3, not 4
END

LESSON 3 1 Decisions and loops
SUMMARY if, else, switch, while, do-while, for, break and continue
# ------------------------------------------------------------
CARD Making decisions
TEXT `if` runs code only when a condition is true. `else if` checks another condition, and `else` catches everything left. Only the first matching branch runs.
TEXT `&&` means AND (both must be true), and `||` means OR (at least one must be true).
CODE int score = 85;
CODE if (score >= 90) {
CODE     printf("Grade: A\n");
CODE } else if (score >= 80) {
CODE     printf("Grade: B\n");    // this one runs
CODE } else {
CODE     printf("Grade: F\n");
CODE }
CODE
CODE if (age >= 18 && age <= 65) printf("Working age\n");
END
CARD The switch statement
TEXT `switch` checks one variable against several exact values. Each `case` needs a `break;` at the end, or the code carries on into the next case (called fall-through).
TEXT Two cases with no code between them share the same code, which is handy for upper and lower case letters. `default` runs when nothing else matches.
CODE switch (grade) {
CODE     case 'A':
CODE     case 'a':
CODE         printf("Excellent!\n");
CODE         break;
CODE     case 'B':
CODE     case 'b':
CODE         printf("Good job!\n");
CODE         break;
CODE     default:
CODE         printf("Invalid grade\n");
CODE }
END
CARD Loops
TEXT A `while` loop checks its condition first, so it may never run. A `do-while` loop checks at the end, so it always runs at least once.
TEXT A `for` loop puts the start, the condition and the step in one line. `break` leaves a loop straight away, and `continue` skips to the next time round.
CODE for (int i = 1; i <= 10; i++) {
CODE     if (i % 2 == 0) {
CODE         continue;   // skip even numbers
CODE     }
CODE     printf("%d ", i);   // 1 3 5 7 9
CODE }
CODE
CODE int number = 3;
CODE do {
CODE     printf("%d ", number);   // 3 2 1
CODE     number--;
CODE } while (number != 0);
END
# ------------------------------------------------------------
QUESTION output
TEXT What does this print?
CODE int score = 75;
CODE if (score >= 90) printf("A");
CODE else if (score >= 80) printf("B");
CODE else if (score >= 70) printf("C");
CODE else printf("F");
OPTION A
OPTION B
OPTION C
OPTION F
ANSWER 3
EXPLAIN 75 isn't >= 90 or >= 80, but it is >= 70, so that branch prints C. Once one branch runs, the rest are skipped.
END
QUESTION output
TEXT What does this print?
CODE for (int i = 1; i <= 6; i++) {
CODE     if (i % 2 == 0) {
CODE         continue;
CODE     }
CODE     printf("%d ", i);
CODE }
OPTION 2 4 6
OPTION 1 3 5
OPTION 1 2 3 4 5 6
OPTION 1 3 5 7
ANSWER 2
EXPLAIN When i is even, continue skips the printf and goes to the next i. So only the odd numbers 1, 3 and 5 are printed (the loop stops after 6).
END
QUESTION choice
TEXT Which loop always runs its body at least once?
OPTION for
OPTION while
OPTION do-while
OPTION None of them
ANSWER 3
EXPLAIN A do-while loop checks its condition at the end, after the body has run once. for and while check first, so their body might never run.
END

LESSON 4 1 Arrays and strings
SUMMARY 1D and 2D arrays, sizeof, strings and the null terminator
# ------------------------------------------------------------
CARD Arrays
TEXT An array stores several values of the same type next to each other in memory. Indexes start at 0, so the last element of a 5-element array is index 4.
TEXT If you give fewer values than the size, the rest are set to 0. `sizeof(arr) / sizeof(arr[0])` works out how many elements an array has.
CODE int numbers[5] = {10, 20, 30, 40, 50};
CODE printf("%d\n", numbers[0]);   // 10 (first)
CODE printf("%d\n", numbers[4]);   // 50 (last)
CODE
CODE int scores[10] = {85, 92, 78};   // the rest are 0
CODE int size = sizeof(numbers) / sizeof(numbers[0]);   // 5
END
CARD 2D arrays
TEXT A 2D array is a grid of rows and columns. `matrix[row][col]` picks one element, and nested loops go through every row and column.
CODE int matrix[3][4] = {
CODE     {1, 2, 3, 4},
CODE     {5, 6, 7, 8},
CODE     {9, 10, 11, 12}
CODE };
CODE printf("%d\n", matrix[2][0]);   // 9
CODE
CODE for (int row = 0; row < 3; row++)
CODE     for (int col = 0; col < 4; col++)
CODE         printf("%3d ", matrix[row][col]);
END
CARD Strings
TEXT C has no separate string type. A string is an array of chars that ends with the null terminator `'\0'`. That's how `printf("%s")` knows where it stops: there's no stored length.
TEXT `char name[20] = "Alice";` stores 'A', 'l', 'i', 'c', 'e', '\0'. If you build the array by hand, you must add the `'\0'` yourself.
CODE char name[20] = "Alice";   // '\0' added for you
CODE
CODE int len = 0;
CODE while (name[len] != '\0') {   // count until the terminator
CODE     len++;
CODE }
CODE printf("%d\n", len);   // 5
END
# ------------------------------------------------------------

OPTION 30
OPTION 40

END

END
